const CLOUD_NAME = "boswge8x"; const UPLOAD_PRESET = "boswge8x"; const uploadButton = document.getElementById("upload-btn"); const previewContainer = document.getElementById("preview-container"); // Enhanced error-tracking widget setup const myWidget = cloudinary.createUploadWidget({ cloudName: CLOUD_NAME, uploadPreset: UPLOAD_PRESET, sources: ['local', 'camera'], multiple: false, maxFileSize: 50000000, clientAllowedFormats: ["png", "jpeg", "jpg", "mp4", "mov"] }, (error, result) => { // If Cloudinary rejects the upload, it will alert you here if (error) { alert("Cloudinary Error: " + error.message); console.error(error); } if (!error && result && result.event === "success") { previewContainer.innerHTML = ''; if (result.info.resource_type === "image") { const img = document.createElement("img"); img.src = result.info.secure_url; previewContainer.appendChild(img); } else if (result.info.resource_type === "video") { const video = document.createElement("video"); video.src = result.info.secure_url; video.controls = true; previewContainer.appendChild(video); } } }); uploadButton.addEventListener("click", function(event) { event.preventDefault(); // Prevents the browser from blocking the button action try { myWidget.open(); } catch(err) { alert("JavaScript Error trying to open widget: " + err.message); } }, false);
